DeepSeek 增强引擎

Ge32 工具箱 / 提供两种部署方式，自由选择

方式一：油猴插件版 (推荐)

特点：一劳永逸，每次打开 DeepSeek 自动加载。

教程： 1. 浏览器安装 Tampermonkey (油猴) 扩展。
2. 点击复制下方代码，在油猴中点击"添加新脚本"，粘贴并保存。
3. 刷新 chat.deepseek.com 页面即可看到控制面板。 

// ==UserScript==
// @name         OrbixMind DeepSeek Harvester (V9.0 - Titanium)
// @namespace    http://tampermonkey.net/
// @version      9.0
// @description  钛金版：保留V8握手协议 + 回滚至V6强力发送内核 + 修复死锁问题
// @author       Orbix Architect
// @match        https://chat.deepseek.com/*
// @grant        GM_addStyle
// @run-at       document-end
// ==/UserScript==

(function() {
    'use strict';

    // ==========================================
    // 1. 配置与状态 (保留 V8 架构)
    // ==========================================
    const CONFIG = {
        templateSuffix: '_tp.txt',
        cooldownMin: 3000,
        cooldownMax: 6000,
        responseTimeout: 300000, // 5分钟超时
        pollInterval: 1000,
    };

    const STATE = {
        isRunning: false,
        queue: [],
        currentIndex: 0,
        directoryHandle: null,
        templateCache: new Map(),
        currentTemplate: "",
        activeRequestId: null, // 握手ID
        sentPromptHash: null   // 自读过滤Hash
    };

    // ==========================================
    // 2. UI 引擎 (Cyberpunk Style)
    // ==========================================
    GM_addStyle(`
        #orbix-dashboard {
            position: fixed !important; top: 50px !important; right: 20px !important; width: 350px !important;
            background: #000 !important; border: 2px solid #00ff00 !important; border-radius: 6px !important;
            padding: 15px !important; z-index: 9999999 !important; font-family: monospace !important;
            box-shadow: 0 0 20px rgba(0, 255, 0, 0.2) !important; color: #fff !important;
        }
        .orbix-btn {
            background: #222; border: 1px solid #444; color: #fff; padding: 6px 12px; cursor: pointer;
            width: 100%; margin-bottom: 5px; font-family: monospace; transition: all 0.2s;
        }
        .orbix-btn:hover { background: #333; border-color: #00ff00; }
        .orbix-primary { background: #004400; border-color: #00ff00; color: #fff; font-weight: bold; }
        .orbix-logger {
            height: 120px; overflow-y: auto; background: #111; border: 1px solid #333;
            font-size: 10px; padding: 5px; margin-top: 10px; color: #0f0; white-space: pre-wrap;
        }
        textarea.orbix-input {
            width: 100%; height: 60px; background: #111; border: 1px solid #333; color: #fff;
            font-family: monospace; padding: 5px; box-sizing: border-box;
        }
    `);

    function initUI() {
        if (document.getElementById('orbix-dashboard')) return;
        const panel = document.createElement('div');
        panel.id = 'orbix-dashboard';
        panel.innerHTML = `
            <div style="font-weight:bold; border-bottom:1px solid #333; margin-bottom:10px; padding-bottom:5px;">
                ORBIX V9.0 // TITANIUM
            </div>

            <button id="orbix-dir-btn" class="orbix-btn">1. 挂载目录</button>

            <div style="display:flex; gap:5px; margin-bottom:5px;">
                <select id="orbix-tp-select" style="flex:1; background:#111; color:#fff; border:1px solid #333;">
                    <option>等待挂载...</option>
                </select>
                <button id="orbix-refresh-btn" class="orbix-btn" style="width:40px; margin:0;">↻</button>
            </div>

            <button id="orbix-new-tp-btn" class="orbix-btn" style="font-size:10px;">📄 新建标准协议模板</button>
            <textarea id="orbix-queue-input" class="orbix-input" placeholder="输入任务队列 (每行一个)..."></textarea>

            <div style="display:flex; gap:5px; margin-top:5px;">
                <button id="orbix-start-btn" class="orbix-btn orbix-primary">▶ 启动</button>
                <button id="orbix-stop-btn" class="orbix-btn">⏹ 停止</button>
            </div>

            <div id="orbix-logger" class="orbix-logger">Waiting for command...</div>
        `;
        document.body.appendChild(panel);

        document.getElementById('orbix-dir-btn').onclick = selectDirectory;
        document.getElementById('orbix-refresh-btn').onclick = refreshTemplates;
        document.getElementById('orbix-new-tp-btn').onclick = createStandardTemplate;
        document.getElementById('orbix-start-btn').onclick = startEngine;
        document.getElementById('orbix-stop-btn').onclick = () => { STATE.isRunning = false; log('🛑 停止'); };
        document.getElementById('orbix-tp-select').onchange = (e) => {
            if (STATE.templateCache.has(e.target.value)) STATE.currentTemplate = STATE.templateCache.get(e.target.value);
        };
    }

    function log(msg) {
        const el = document.getElementById('orbix-logger');
        if (el) {
            const time = new Date().toLocaleTimeString().split(' ')[0];
            el.innerHTML += `<div>[${time}] ${msg}</div>`;
            el.scrollTop = el.scrollHeight;
        }
    }

    // ==========================================
    // 3. 文件系统 (V8逻辑)
    // ==========================================
    async function selectDirectory() {
        try {
            STATE.directoryHandle = await window.showDirectoryPicker();
            log(`✅ 目录: ${STATE.directoryHandle.name}`);
            await refreshTemplates();
        } catch (e) { log(`❌ ${e.message}`); }
    }

    async function refreshTemplates() {
        if (!STATE.directoryHandle) return;
        const select = document.getElementById('orbix-tp-select');
        select.innerHTML = '<option>-- 选择模板 --</option>';
        STATE.templateCache.clear();
        for await (const entry of STATE.directoryHandle.values()) {
            if (entry.name.endsWith(CONFIG.templateSuffix)) {
                const f = await (await STATE.directoryHandle.getFileHandle(entry.name)).getFile();
                STATE.templateCache.set(entry.name, await f.text());
                const opt = document.createElement('option');
                opt.value = entry.name;
                opt.textContent = entry.name;
                select.appendChild(opt);
            }
        }
        log('📋 模板刷新完毕');
    }

    async function createStandardTemplate() {
        if (!STATE.directoryHandle) return alert("先挂载目录");
        const content = `Topic: "{{topic}}"

REQUIREMENTS:
1. Output in Markdown.
2. WRAP OUTPUT IN <file> TAGS.
3. INCLUDE THE ID PROVIDED IN THE TAG.

<file name="{{topic}}_res.md">
# {{topic}}
[Content...]
</file>`;
        try {
            const h = await STATE.directoryHandle.getFileHandle(`V9_Standard${CONFIG.templateSuffix}`, { create: true });
            const w = await h.createWritable();
            await w.write(content);
            await w.close();
            await refreshTemplates();
            log('✅ 标准模板已创建');
        } catch (e) { log(e.message); }
    }

    // ==========================================
    // 4. 核心引擎 (Titanium Core)
    // ==========================================

    // 工具：生成ID
    const genId = () => 'req_' + Math.random().toString(36).substr(2, 6);
    // 工具：Hash
    const calcHash = (s) => s.split('').reduce((a, b) => { a = ((a << 5) - a) + b.charCodeAt(0); return a & a }, 0);
    // 工具：注入ID
    const injectId = (p, id) => p.includes('<file') ? p.replace(/<file/i, `<file id="${id}"`) : p + `\n\n<file id="${id}" name="out.md">...</file>`;

    async function startEngine() {
        if (!STATE.directoryHandle) return alert("未挂载目录");
        const input = document.getElementById('orbix-queue-input').value.trim();
        if (!input) return alert("队列为空");

        STATE.queue = input.split('\n').map(s => s.trim()).filter(s => s);
        STATE.currentIndex = 0;
        STATE.isRunning = true;

        if (!STATE.currentTemplate) STATE.currentTemplate = `Write about {{topic}}. <file>Content</file>`;

        processNext();
    }

    async function processNext() {
        if (!STATE.isRunning) return;
        if (STATE.currentIndex >= STATE.queue.length) { STATE.isRunning = false; log("🎉 完成"); return; }

        const topic = STATE.queue[STATE.currentIndex];
        log(`\n>>> [${STATE.currentIndex+1}/${STATE.queue.length}] ${topic}`);

        // 1. 准备数据
        STATE.activeRequestId = genId();
        let prompt = STATE.currentTemplate.replace(/{{topic}}/g, topic);
        prompt = injectId(prompt, STATE.activeRequestId);
        STATE.sentPromptHash = calcHash(prompt);

        try {
            // 2. 强力发送
            await titaniumSend(prompt);

            // 3. 握手等待
            log("⏳ 等待握手响应...");
            const result = await handshakeWait();

            // 4. 保存
            if (result) {
                await saveToDisk(result.filename, result.content);
                log(`💾 已保存: ${result.filename}`);
            }

            // 5. 冷却
            STATE.currentIndex++;
            const cd = Math.floor(Math.random() * (CONFIG.cooldownMax - CONFIG.cooldownMin) + CONFIG.cooldownMin);
            log(`❄️ 冷却 ${(cd/1000).toFixed(1)}s...`);
            await sleep(cd);
            processNext();

        } catch (e) {
            log(`❌ ${e.message}`);
            STATE.currentIndex++;
            setTimeout(processNext, 3000);
        }
    }

    // --- 模块 A: 钛金发送 ---
    async function titaniumSend(text) {
        const MAX_RETRIES = 5;
        let retries = 0;

        while (retries < MAX_RETRIES && STATE.isRunning) {
            const textarea = document.querySelector('textarea');
            if (!textarea) throw new Error("找不到输入框");

            textarea.focus();
            const nativeSetter = Object.getOwnPropertyDescriptor(window.HTMLTextAreaElement.prototype, "value").set;
            nativeSetter.call(textarea, text);

            const events = ['input', 'change', 'keydown', 'keypress', 'keyup'];
            events.forEach(evt => textarea.dispatchEvent(new Event(evt, { bubbles: true })));

            await sleep(1000);

            const btns = Array.from(document.querySelectorAll('button, div[role="button"]'));
            const sendBtn = btns.find(b => {
                const html = b.innerHTML.toLowerCase();
                const label = (b.getAttribute('aria-label') || '').toLowerCase();
                const active = !b.disabled && b.getAttribute('aria-disabled') !== 'true';
                const match = label.includes('send') || label.includes('发送') || html.includes('path') || html.includes('svg');
                return active && match && b.clientWidth < 100;
            });

            if (sendBtn) {
                sendBtn.click();
            } else {
                textarea.dispatchEvent(new KeyboardEvent('keydown', { key: 'Enter', code: 'Enter', keyCode: 13, bubbles: true }));
            }

            await sleep(1500); 
            if (textarea.value.trim() === '') {
                log("🚀 发送成功");
                return true;
            } else {
                log(`⚠️ 发送未响应 (尝试 ${retries+1}/${MAX_RETRIES})`);
                textarea.dispatchEvent(new KeyboardEvent('keydown', { key: 'Enter', code: 'Enter', keyCode: 13, bubbles: true }));
                await sleep(1000);
                if (textarea.value.trim() === '') return true;
            }

            retries++;
        }
        throw new Error("发送失败：输入框死锁");
    }

    // --- 模块 B: 握手等待 ---
    async function handshakeWait() {
        const start = Date.now();
        while (Date.now() - start < CONFIG.responseTimeout && STATE.isRunning) {
            await sleep(1000);

            const msgs = document.querySelectorAll('.ds-markdown, .markdown-body');
            if (msgs.length === 0) continue;

            for (let i = msgs.length - 1; i >= 0; i--) {
                const text = msgs[i].innerText;

                if (!text.includes(STATE.activeRequestId)) continue;
                if (!text.includes('</file>')) continue;
                if (calcHash(text) === STATE.sentPromptHash) continue;

                const fileData = extractFile(text);
                if (fileData) return fileData;
            }
        }
        throw new Error("响应超时");
    }

    // --- 模块 C: 文件与工具 ---
    function extractFile(text) {
        const id = STATE.activeRequestId;
        const regex = new RegExp(`<file[^>]*id="${id}"[^>]*>([\\s\\S]*?)<\\/file>`, 'i');
        const match = regex.exec(text);
        if (match) {
            const nameMatch = text.match(/name="([^"]+)"/);
            return {
                filename: nameMatch ? nameMatch[1] : `${id}.md`,
                content: match[1].trim()
            };
        }
        return null;
    }

    async function saveToDisk(name, content) {
        if (!STATE.directoryHandle) return;
        try {
            const safeName = name.replace(/[\\/:*?"<>|]/g, '_');
            const h = await STATE.directoryHandle.getFileHandle(safeName, { create: true });
            const w = await h.createWritable();
            await w.write(content);
            await w.close();
        } catch(e) { log(`保存错: ${e.message}`); }
    }

    function sleep(ms) { return new Promise(r => setTimeout(r, ms)); }

    setTimeout(initUI, 1500);
})();

方式二：控制台极速版 (免安装)

特点：即插即用，无需安装任何扩展，随取随用。

教程： 1. 打开并登录 chat.deepseek.com。
2. 键盘按下 F12 (Mac 快捷键 Cmd+Opt+J) 打开开发者工具。
3. 切换到 Console (控制台) 标签，粘贴下方代码并按回车运行。

    // 1. 配置与状态
    // ==========================================

        cooldownMin: 3000,
        cooldownMax: 6000,

    function log(msg) {
        const el = document.getElementById('ge32-logger');
        if (el) {
            const time = new Date().toLocaleTimeString().split(' ')[0];
            el.innerHTML += `<div>[${time}] ${msg}</div>`;
            el.scrollTop = el.scrollHeight;
        }
    }

    // ==========================================
    // 3. 核心引擎
    // ==========================================
    const sleep = (ms) => new Promise(r => setTimeout(r, ms));

    function getMessages() { return document.querySelectorAll('.ds-markdown, .markdown-body'); }

    async function waitForMessageCount(targetCount, timeout = 30000) {
        const start = Date.now();
        while (Date.now() - start < timeout) {
            if (getMessages().length >= targetCount) return true;
            await sleep(CONFIG.checkInterval);
        }
        return false;
    }

    async function waitForContentStable(messageElement, timeout = CONFIG.responseTimeout) {
        let lastContent = '';
        let stableCount = 0;
        const start = Date.now();

        while (Date.now() - start < timeout) {
            const current = messageElement.innerText;
            if (current === lastContent) {
                stableCount++;
                if (stableCount >= CONFIG.stabilityChecks) return current;
            } else {
                stableCount = 0;
                lastContent = current;
            }
            await sleep(800);
        }
        return messageElement.innerText;
    }

    async function sendPrompt(text) {
        const MAX_RETRIES = 5;
        let retries = 0;

        while (retries < MAX_RETRIES && STATE.isRunning) {
            const textarea = document.querySelector('textarea');
            if (!textarea) throw new Error('找不到输入框');

            textarea.focus();
            const nativeSetter = Object.getOwnPropertyDescriptor(window.HTMLTextAreaElement.prototype, 'value').set;
            nativeSetter.call(textarea, text);

            ['input', 'change', 'keydown', 'keypress', 'keyup'].forEach(evt =>
                textarea.dispatchEvent(new Event(evt, { bubbles: true }))
            );

            await sleep(1000);

            const btns = Array.from(document.querySelectorAll('button, div[role="button"]'));
            const sendBtn = btns.find(b => {
                const html = b.innerHTML.toLowerCase();
                const label = (b.getAttribute('aria-label') || '').toLowerCase();
                const active = !b.disabled && b.getAttribute('aria-disabled') !== 'true';
                const match = label.includes('send') || label.includes('发送') || html.includes('path') || html.includes('svg');
                return active && match && b.clientWidth < 100;

            if (sendBtn) {
                sendBtn.click();
            } else {
                textarea.dispatchEvent(new KeyboardEvent('keydown', { key: 'Enter', code: 'Enter', keyCode: 13, bubbles: true }));

            if (textarea.value.trim() === '') {
                log('🚀 发送成功');
                return true;
            } else {
                log(`⚠️ 发送未响应 (尝试 ${retries+1}/${MAX_RETRIES})`);
                textarea.dispatchEvent(new KeyboardEvent('keydown', { key: 'Enter', code: 'Enter', keyCode: 13, bubbles: true }));
                await sleep(1000);
                if (textarea.value.trim() === '') return true;
            }
            retries++;
        }
        throw new Error('发送失败：输入框死锁');
    }

    async function waitForResponse() {
        const initialCount = getMessages().length;
        const targetUserCount = initialCount + 1;

        if (!(await waitForMessageCount(targetUserCount, 30000))) throw new Error('用户消息未出现');

        const targetAICount = targetUserCount + 1;
        if (!(await waitForMessageCount(targetAICount, CONFIG.responseTimeout))) throw new Error('AI响应超时');

        const messages = getMessages();
        const latestMsg = messages[messages.length - 1];

        return await waitForContentStable(latestMsg, CONFIG.responseTimeout);
    }

    async function processNext() {
        if (!STATE.isRunning) return;
        if (STATE.currentIndex >= STATE.queue.length) {
            STATE.isRunning = false;
            log('🎉 所有任务完成');
            return;
        }

        const prompt = STATE.queue[STATE.currentIndex];
        log(`\n>>> [${STATE.currentIndex+1}/${STATE.queue.length}] 发送...`);

        try {
            await sendPrompt(prompt);
            log('⏳ 等待 AI 响应...');
            const response = await waitForResponse();
            log(`💬 收到响应 (${response.length} 字符)`);
            log(`📄 预览: ${response.substring(0, 100)}...`);

            STATE.currentIndex++;
            const cd = Math.floor(Math.random() * (CONFIG.cooldownMax - CONFIG.cooldownMin) + CONFIG.cooldownMin);
            log(`❄️ 冷却 ${(cd/1000).toFixed(1)}s...`);
            await sleep(cd);
            processNext();
        } catch (err) {
            log(`❌ 错误: ${err.message}`);
            STATE.currentIndex++;
            await sleep(3000);
            processNext();
        }
    }

    function startEngine() {
        const raw = document.getElementById('ge32-queue-input').value.trim();
        if (!raw) { log('❌ 请输入JSON任务列表'); return; }

        try {
            const taskArray = JSON.parse(raw);
            if (!Array.isArray(taskArray)) throw new Error('不是数组');
            STATE.queue = taskArray.filter(item => typeof item === 'string' && item.trim() !== '');
        } catch (e) {
            log('❌ JSON 解析失败，格式示例: ["task1", "task2"]');
            return;
        }

        STATE.currentIndex = 0;
        STATE.isRunning = true;
        log(`🚀 启动，共 ${STATE.queue.length} 个任务`);
        processNext();
    }

    setTimeout(createUI, 1500);
})();
